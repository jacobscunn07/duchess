```go
// renderFooter returns the footer string for the current model state.
// Left: IAM identity (state-dependent). Right: live clock.
// The result is exactly m.width display columns wide. Returns "" when m.width == 0.
func renderFooter(m rootModel) string {
    if m.width == 0 {
        return ""
    }

    // Left: IAM identity — state-dependent
    var leftStr string
    switch m.state {
    case stateLoading:
        leftStr = identityStyle.Render(m.spinner.View() + " loading...")
    case stateError:
        leftStr = identityStyle.Render("error")
    case stateReady:
        leftStr = identityStyle.Render(m.account + " " + truncateARN(m.arn, 40))
    }

    // Right: live clock
    rightStr := clockStyle.Render(m.now.Format("15:04:05"))

    // Gap fill to full width
    gap := m.width - lipgloss.Width(leftStr) - lipgloss.Width(rightStr)
    if gap < 0 {
        gap = 0
    }

    return statusBarStyle.Width(m.width).Render(leftStr + strings.Repeat(" ", gap) + rightStr)
}
```

Remove the `fmt` import if it is no longer used after stripping the `fmt.Sprintf` call in the old leftStr. The `strings` import stays (used by `strings.Repeat`). Keep `truncateARN` unchanged — it is still used.
