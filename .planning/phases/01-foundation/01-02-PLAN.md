NewAWSConfig(ctx context.Context, profile, region string) (aws.Config, error):
  - Build opts slice: []func(*config.LoadOptions) error{}
  - If profile != "", append config.WithSharedConfigProfile(profile)
  - If region != "", append config.WithRegion(region)
  - Call cfg, err := config.LoadDefaultConfig(ctx, opts...)
  - If err != nil, return aws.Config{}, ClassifyCredentialError(err, profile)
  - After successful load, if cfg.Region == "" return aws.Config{}, fmt.Errorf("no region configured for profile %q — use --region flag or set region in ~/.duchess/config", profile)
  - Return cfg, nil

GetCallerIdentity(ctx context.Context, cfg aws.Config) (account, arn string, err error):
  - client := sts.NewFromConfig(cfg)
  - out, err := client.GetCallerIdentity(ctx, &sts.GetCallerIdentityInput{})
  - If err != nil, return "", "", err
  - Return aws.ToString(out.Account), aws.ToString(out.Arn), nil

ClassifyCredentialError(err error, profile string) error:
  - If err == nil return nil
  - Check ssocreds.InvalidTokenError FIRST (before other handlers):
    var invalidToken *ssocreds.InvalidTokenError
    if errors.As(err, &invalidToken) {
        return fmt.Errorf("SSO session expired. Run: aws sso login --profile %s", profile)
    }
  - Check smithy.APIError for specific error codes:
    var apiErr smithy.APIError
    if errors.As(err, &apiErr) {
        switch apiErr.ErrorCode() {
        case "ExpiredTokenException":
            return fmt.Errorf("credentials expired for profile %q. Refresh your credentials and try again.", profile)
        case "InvalidClientTokenId", "AuthFailure":
            return fmt.Errorf("invalid credentials for profile %q. Check your access key and secret key.", profile)
        case "AccessDenied":
            return fmt.Errorf("access denied for profile %q. Check IAM permissions.", profile)
        }
    }
  - Fall through: return fmt.Errorf("authentication failed for profile %q: %w", profile, err)
