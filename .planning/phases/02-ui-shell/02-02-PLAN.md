**Step 1 — Update internal/ui/model.go View() and add contentView():**

Replace the placeholder `View()` from Plan 02-01 with the full layout:

```go
func (m rootModel) View() string {
    if m.width == 0 {
        return ""
    }
    statusBar := renderStatusBar(m)
    contentH := m.height - lipgloss.Height(statusBar)
    if contentH < 0 {
        contentH = 0
    }
    content := lipgloss.NewStyle().
        Width(m.width).
        Height(contentH).
        Render(m.contentView())
    return lipgloss.JoinVertical(lipgloss.Left, content, statusBar)
}
```

Add `contentView() string` method on rootModel:
