// clusterToItem converts an ECS Cluster to an ecsItem for the list.
func clusterToItem(c ecstypes.Cluster) ecsItem {
    name := aws.ToString(c.ClusterName)
    status := aws.ToString(c.Status)
    return ecsItem{
        kind:    kindCluster,
        left:    name,
        right:   status,
        cluster: &c,
    }
}

// serviceToItem converts an ECS Service to an ecsItem.
func serviceToItem(svc ecstypes.Service) ecsItem {
    name := aws.ToString(svc.ServiceName)
    tag := ""
    switch svc.LaunchType {
    case ecstypes.LaunchTypeFargate:
        tag = "[FARGATE]"
    case ecstypes.LaunchTypeEc2:
        tag = "[EC2]"
    }
    left := name
    if tag != "" {
        left = name + "  " + tag
    }

    right := fmt.Sprintf("%d/%d", svc.RunningCount, svc.DesiredCount)
    if svc.PendingCount > 0 {
        right += fmt.Sprintf(" (%d pending)", svc.PendingCount)
    }

    return ecsItem{
        kind:    kindService,
        left:    left,
        right:   right,
        service: &svc,
    }
}

// taskToItem converts an ECS Task to an ecsItem.
// Short ID = last path segment of TaskArn, truncated to 8 chars.
func taskToItem(t ecstypes.Task) ecsItem {
    shortID := taskShortDisplay(aws.ToString(t.TaskArn))
    status := aws.ToString(t.LastStatus)

    age := ""
    if t.StartedAt != nil {
        age = humanize.Time(*t.StartedAt)
    }

    right := status
    if age != "" {
        right = status + "  " + age
    }

    return ecsItem{
        kind:  kindTask,
        left:  shortID,
        right: right,
        task:  &t,
    }
}

**CRITICAL API CONSTRAINTS (from research/pitfall analysis):**
- `ListClusters` paginator: `ecs.NewListClustersPaginator` — returns ARNs, must follow with `DescribeClusters` (batch max: no hard limit documented, but use 100 to be safe)
- `ListServices` paginator: `ecs.NewListServicesPaginator` — returns ARNs; follow with `DescribeServices` in **chunks of 10** (hard API limit)
- `ListTasks` paginator: `ecs.NewListTasksPaginator` — requires `ServiceName` as short name (NOT service ARN); extract: `strings.LastIndex(serviceArn, "/")+1`; follow with `DescribeTasks` in **chunks of 100** (hard API limit)
- `DescribeTaskDefinition`: takes `taskDefinitionArn` string; log configuration lives on `ContainerDefinition`, NOT on `Container` from DescribeTasks

**FetchServicesCmd**: Calls `ListAllServices(ctx, client, clusterArn)`, returns `servicesLoadedMsg{services}` or `servicesErrMsg{err}`.

**FetchTasksCmd**: Calls `ListAndDescribeTasks(ctx, client, clusterArn, serviceArn)`, returns `tasksLoadedMsg{tasks}` or `tasksErrMsg{err}`.

1. `client.DescribeTasks(ctx, &awsecs.DescribeTasksInput{Cluster: aws.String(clusterArn), Tasks: []string{taskArn}})` — gets the single task

// taskShortID extracts the UUID portion from a task ARN.
// Task ARN format: arn:aws:ecs:region:account:task/cluster-name/uuid OR arn:aws:ecs:...:task/uuid
func taskShortID(taskArn string) string {
    idx := strings.LastIndex(taskArn, "/")
    if idx < 0 || idx >= len(taskArn)-1 {
        return taskArn
    }
    return taskArn[idx+1:]
}

// taskShortDisplay returns the first 8 chars of the task short UUID (like git short SHA).
func taskShortDisplay(taskArn string) string {
    id := taskShortID(taskArn)
    if len(id) > 8 {
        return id[:8]
    }
    return id
}

// buildCloudWatchURL constructs the CloudWatch Logs console deep-link URL.
// CloudWatch deep-link format uses non-standard encoding: / becomes $252F.
// Use: PathEscape (converts / to %2F), then replace % with $25 (giving $252F for /).
func buildCloudWatchURL(region, logGroup, logStream string) string {
    cwEncode := func(s string) string {
        escaped := url.PathEscape(s)
        return strings.ReplaceAll(escaped, "%", "$25")
    }
    return fmt.Sprintf(
        "https://%s.console.aws.amazon.com/cloudwatch/home?region=%s#logsV2:log-groups/log-group/%s/log-events/%s",
        region, region,
        cwEncode(logGroup),
        cwEncode(logStream),
    )
}

// openURL opens a URL in the system browser.
// Uses "open" on macOS (darwin) and "xdg-open" on Linux.
func openURL(rawURL string) error {
    var cmd *exec.Cmd
    switch runtime.GOOS {
    case "darwin":
        cmd = exec.Command("open", rawURL)
    default:
        cmd = exec.Command("xdg-open", rawURL)
    }
    return cmd.Start()
}

// logURLForContainer builds the CloudWatch Logs URL for a specific container.
// logConfig is from the ContainerDefinition (task definition), NOT the Container (task).
// logStream format: {prefix}/{containerName}/{taskShortID}
// Returns "" and an error string if no awslogs driver is configured.
func logURLForContainer(region, taskArn string, containerName string, logConfig *ecstypes.LogConfiguration) (string, string) {
    if logConfig == nil || logConfig.LogDriver != ecstypes.LogDriverAwslogs {
        return "", "No CloudWatch logs configured for this container"
    }
    logGroup := logConfig.Options["awslogs-group"]
    prefix := logConfig.Options["awslogs-stream-prefix"]
    if logGroup == "" {
        return "", "No CloudWatch logs configured for this container"
    }
    shortID := taskShortID(taskArn)
    logStream := prefix + "/" + containerName + "/" + shortID
    return buildCloudWatchURL(region, logGroup, logStream), ""
}
